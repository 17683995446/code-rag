# 演示代码
from kg_memory.core import MemorySystem
ms = MemorySystem()

# 记住知识
ms.remember("Python")
ms.remember("FastAPI")
ms.remember("NetworkX")

# 建立连接
ms.connect("FastAPI", "Python", "uses")
ms.connect("NetworkX", "Python", "uses")

# 分析
pagerank = ms.pagerank()
communities = ms.find_communities()
similar = ms.get_similar_nodes("Python")
